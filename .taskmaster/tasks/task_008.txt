Implement performance monitoring functions that only run when PERFORMANCE_MONITORING is enabled:

```typescript
interface PerformanceMetrics {
  frameTime: number[];
  memoryUsage?: any;
  lastReportTime: number;
}

const metrics: PerformanceMetrics = {
  frameTime: [],
  lastReportTime: 0
};

function trackPerformance(startTime: number): void {
  if (!CONFIG.PERFORMANCE_MONITORING) return;
  
  // Track frame time
  const frameTime = performance.now() - startTime;
  metrics.frameTime.push(frameTime);
  
  // Limit array size
  if (metrics.frameTime.length > 100) {
    metrics.frameTime.shift();
  }
  
  // Track memory usage if available
  if (performance.memory) {
    metrics.memoryUsage = performance.memory;
  }
  
  // Report metrics every 5 seconds
  const now = performance.now();
  if (now - metrics.lastReportTime > 5000) {
    reportPerformanceMetrics();
    metrics.lastReportTime = now;
  }
}

function reportPerformanceMetrics(): void {
  if (!CONFIG.PERFORMANCE_MONITORING) return;
  
  // Calculate average frame time
  const avgFrameTime = metrics.frameTime.reduce((sum, time) => sum + time, 0) / metrics.frameTime.length;
  
  debug('Performance metrics:');
  debug(`- Average frame time: ${avgFrameTime.toFixed(2)}ms`);
  debug(`- Max frame time: ${Math.max(...metrics.frameTime).toFixed(2)}ms`);
  
  if (metrics.memoryUsage) {
    debug(`- Memory usage: ${(metrics.memoryUsage.usedJSHeapSize / 1024 / 1024).toFixed(2)}MB`);
  }
}
```

Modify the scramble loop to include performance tracking:

```typescript
function scrambleLoop(timestamp: number): void {
  if (!isRunning) return;
  
  // Request next frame first
  requestAnimationFrame(scrambleLoop);
  
  // Check if it's time to scramble
  if (timestamp - lastScrambleTime >= CONFIG.SCRAMBLE_INTERVAL) {
    const startTime = performance.now();
    lastScrambleTime = timestamp;
    
    // Perform scrambling
    scrambleAllClasses(registry);
    
    // Track performance
    trackPerformance(startTime);
    
    // Periodically clean up registry
    if (Math.floor(timestamp / 1000) % CONFIG.CLEANUP_INTERVAL === 0) {
      cleanupRegistry(registry);
    }
  }
}
```
