Implement functions to generate random class names and check for conflicts:

```typescript
function generateRandomClassName(length = 1): string {
  let result = '';
  const charset = CONFIG.CHARSET;
  
  // First character must be a letter (a-z)
  result += charset.substring(0, 26).charAt(Math.floor(Math.random() * 26));
  
  // If length > 1, add more random characters (can include numbers)
  for (let i = 1; i < length; i++) {
    result += charset.charAt(Math.floor(Math.random() * charset.length));
  }
  
  return result;
}

function generateUniqueClassName(registry: ClassRegistry, preferredLength = 1): string {
  // Start with preferred length, increase if needed to find unique name
  let length = preferredLength;
  let attempts = 0;
  let className;
  
  do {
    className = generateRandomClassName(length);
    attempts++;
    
    // If we've tried too many times at this length, increase length
    if (attempts > 10) {
      length++;
      attempts = 0;
    }
  } while (registry[className]);
  
  return className;
}
```

Implement the function to scramble a single class:

```typescript
function scrambleClass(registry: ClassRegistry, originalClassName: string): void {
  const entry = registry[originalClassName];
  if (!entry) return;
  
  try {
    // Generate a new unique class name
    const newClassName = generateUniqueClassName(registry, originalClassName.length);
    
    // Update the CSS rule
    entry.cssRule.selectorText = `.${newClassName}`;
    
    // Update all DOM elements
    for (const element of entry.elements) {
      element.classList.remove(originalClassName);
      element.classList.add(newClassName);
    }
    
    // Update registry with new class name
    registry[newClassName] = entry;
    delete registry[originalClassName];
  } catch (e) {
    debug(`Error scrambling class ${originalClassName}`, e);
  }
}
```
