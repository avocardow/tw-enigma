# Description: Implement a system to map CSS rules to their corresponding DOM elements and create a registry for synchronized updates.
# Details:
Create a registry data structure to map CSS rules to DOM elements:

```typescript
interface ClassRegistry {
  [className: string]: {
    cssRule: CSSStyleRule;
    elements: Element[];
    originalClassName: string;
  }
}

async function buildRegistry(): Promise<ClassRegistry> {
  const registry: ClassRegistry = {};
  const selectors = await discoverWithRetry();
  
  debug(`Discovered ${selectors.length} tw-enigma selectors`);
  
  // Limit registry size to prevent performance issues
  const limitedSelectors = selectors.slice(0, CONFIG.MAX_REGISTRY_SIZE);
  
  for (const { rule, className } of limitedSelectors) {
    try {
      // Find all elements with this class
      const elements = Array.from(document.querySelectorAll(`.${className}`));
      
      if (elements.length > 0) {
        registry[className] = {
          cssRule: rule,
          elements,
          originalClassName: className
        };
      }
    } catch (e) {
      debug(`Error mapping elements for class ${className}`, e);
    }
  }
  
  debug(`Built registry with ${Object.keys(registry).length} active classes`);
  return registry;
}
```

Implement a registry cleanup function to prevent memory leaks:

```typescript
function cleanupRegistry(registry: ClassRegistry): void {
  for (const className in registry) {
    // Filter out elements that are no longer in the DOM
    registry[className].elements = registry[className].elements.filter(el => el.isConnected);
    
    // Remove classes with no elements
    if (registry[className].elements.length === 0) {
      delete registry[className];
    }
  }
  
  debug(`Registry cleanup complete, ${Object.keys(registry).length} classes remaining`);
}
```
