# Description: Add support for respecting the prefers-reduced-motion media query to automatically disable the scrambling effect for users with motion sensitivity.
# Details:
Implement a function to check for reduced motion preference:

```typescript
function shouldReduceMotion(): boolean {
  return window.matchMedia('(prefers-reduced-motion: reduce)').matches;
}
```

Modify the initialization function to respect this preference:

```typescript
async function initialize(): Promise<void> {
  // Check for reduced motion preference
  if (shouldReduceMotion()) {
    debug('Reduced motion preference detected, scrambling disabled');
    return;
  }
  
  try {
    // Build the registry
    registry = await buildRegistry();
    
    // Start scrambling if we found any classes
    if (Object.keys(registry).length > 0) {
      startScrambling();
    } else {
      debug('No tw-enigma classes found, scrambling disabled');
    }
  } catch (e) {
    debug('Error initializing scramble effect', e);
  }
}
```

Add a listener for changes to the media query:

```typescript
function setupMotionPreferenceListener(): void {
  const mediaQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
  
  // Modern browsers
  if (mediaQuery.addEventListener) {
    mediaQuery.addEventListener('change', () => {
      if (mediaQuery.matches) {
        stopScrambling();
      } else {
        initialize();
      }
    });
  }
  // Older browsers
  else if (mediaQuery.addListener) {
    mediaQuery.addListener(() => {
      if (mediaQuery.matches) {
        stopScrambling();
      } else {
        initialize();
      }
    });
  }
}
```
