```typescript
let lastScrambleTime = 0;
let registry: ClassRegistry = {};
let isRunning = false;

function scrambleLoop(timestamp: number): void {
  if (!isRunning) return;
  
  // Request next frame first to ensure smooth animation
  requestAnimationFrame(scrambleLoop);
  
  // Check if it's time to scramble
  if (timestamp - lastScrambleTime >= CONFIG.SCRAMBLE_INTERVAL) {
    lastScrambleTime = timestamp;
    
    // Perform scrambling
    scrambleAllClasses(registry);
    
    // Periodically clean up registry (every CLEANUP_INTERVAL seconds)
    if (Math.floor(timestamp / 1000) % CONFIG.CLEANUP_INTERVAL === 0) {
      cleanupRegistry(registry);
    }
  }
}

function startScrambling(): void {
  if (isRunning) return;
  isRunning = true;
  requestAnimationFrame(scrambleLoop);
  debug('Scrambling started');
}

function stopScrambling(): void {
  isRunning = false;
  debug('Scrambling stopped');
}
```
